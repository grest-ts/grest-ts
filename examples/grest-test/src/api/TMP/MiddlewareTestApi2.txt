GG_LANGUAGE.useHttpHeader({
    type: IsObject({
        'accept-language': IsString
    }),
    decode: (value) => {
        const lang = value["accept-language"]
        return {
            language: lang && lang !== '*' && lang.trim() !== '' ? lang.split(',')[0].split('-')[0] : undefined
        }
    },
    encode: (value) => {
        return {
            "accept-language": value.language
        }
    }
})

export interface ClientInfo {
    version: string;
    platform: 'web' | 'ios' | 'android';
}

export const GG_CLIENT_INFO = GGContextKey.create<ClientInfo>('clientInfo');

GG_CLIENT_INFO.use(GGHttp.createTransport({
    updateRequest(req: GGHttpRequest): void {
        const info = GG_CLIENT_INFO.get();
        if (info) {
            req.headers['x-client-version'] = info.version;
            req.headers['x-client-platform'] = info.platform;
        }
    },
    parseRequest(req: GGHttpRequest): void {
        GG_CLIENT_INFO.set({
            version: req.headers['x-client-version'] ?? 'unknown',
            platform: (req.headers['x-client-platform'] ?? 'web') as 'web' | 'ios' | 'android'
        });
    }
}));

// -------------------------------------------------------------------------------

export interface FeatureFlags {
    darkMode: boolean;
    betaFeatures: boolean;
}

export const GG_FEATURE_FLAGS = GGContextKey.create<FeatureFlags>('features');

GG_FEATURE_FLAGS.use(GGHttp.createTransport({
    updateRequest(req: GGHttpRequest): void {
        const val = GG_FEATURE_FLAGS.get();
        req.headers['x-feature-flags'] = val ? JSON.stringify(val) : "";
    },
    parseRequest(req: GGHttpRequest): void {
        const data = req.headers['x-feature-flags'];
        const raw = data ? JSON.parse(data) : {};
        GG_FEATURE_FLAGS.set({
            darkMode: raw.darkMode ?? false,
            betaFeatures: raw.betaFeatures ?? false
        });
    }
}));

export type MiddlewareTestRequest = typeof IsMiddlewareTestRequest.infer
export type MiddlewareTestResponse = typeof IsMiddlewareTestResponse.infer
export type LanguageResponse = typeof IsLanguageResponse.infer

// SHARABLE DEFINITION
const MiddlewareTestApiContract = gg.contract("MiddlewareTestApi")
    .use(GG_LANGUAGE)
    .use(GG_CLIENT_INFO)
    .use(GG_FEATURE_FLAGS)
    .methods({
        echo: {
            input: IsMiddlewareTestRequest,
            success: IsMiddlewareTestResponse,
            errors: [VALIDATION_ERROR, SERVER_ERROR]
        },
        getLanguage: {
            success: IsLanguageResponse,
            errors: [SERVER_ERROR]
        }
    })

// Implementation
class MiddlewareTestApiContractImpl implements GGServerApi<typeof MiddlewareTestApiContract.infer> {
    async echo(request: MiddlewareTestRequest): Promise<MiddlewareTestResponse> {
        const language = GG_LANGUAGE.get();
        const clientInfo = GG_CLIENT_INFO.get();
        const features = GG_FEATURE_FLAGS.get();
        return {
            message: request.message,
            language: language,
            clientVersion: clientInfo.version,
            clientPlatform: clientInfo.platform,
            darkMode: features.darkMode,
            betaFeatures: features.betaFeatures
        }
    }

    async getLanguage(): Promise<LanguageResponse> {
        return Promise.resolve({language: GG_LANGUAGE.get()})
    }
}

// Http mapping
export const MiddlewareTestApiHttpMapping = gg.http
    .for(MiddlewareTestApiContract.create()
        .use(USER_AUTH)
        .impl(new MiddlewareTestApiContractImpl())
    )
    .pathPrefix("api/middleware-test")
    .routes({
        echo: gg.http.POST("echo"),
        getLanguage: gg.http.GET("language")
    })
